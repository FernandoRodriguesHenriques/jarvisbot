function atualizarTimerVela(){let agora = new Date();let segundos = agora.getSeconds();let minutos = agora.getMinutes();let horas = agora.getHours();let restoSeg = 0;let totalSeg = 0;if(TF==="M1"){ totalSeg=60; restoSeg=60-segundos; }else if(TF==="M5"){ totalSeg=300; restoSeg= (5 - (minutos%5) -1)*60 + (60-segundos); }else if(TF==="M15"){ totalSeg=900; restoSeg= (15 - (minutos%15) -1)*60 + (60-segundos); }else if(TF==="M30"){ totalSeg=1800; restoSeg= (30 - (minutos%30) -1)*60 + (60-segundos); }else if(TF==="H1"){ totalSeg=3600; restoSeg= (60 - minutos -1)*60 + (60-segundos); }else if(TF==="H4"){ totalSeg=14400; let bloco = Math.floor(horas/4)*4; restoSeg= ( (bloco+4)*3600 - (horas*3600 + minutos*60 + segundos) ); }else if(TF==="D1"){ totalSeg=86400; restoSeg= 86400 - (horas*3600 + minutos*60 + segundos); }else { totalSeg=60; restoSeg=60-segundos; }if(restoSeg<=0) restoSeg=0;let m = Math.floor(restoSeg/60);let s = restoSeg%60;let timerEl = document.getElementById('candleTimer');timerEl.textContent = `${m.toString().padStart(2,'0')}:${s.toString().padStart(2,'0')}`;if(restoSeg <= 10){timerEl.style.color="#ff3d57";}else if(restoSeg <= 60){timerEl.style.color="#ffeb3b";}else{timerEl.style.color="#00e676";}document.getElementById('candleClock').textContent = agora.toLocaleTimeString('pt-BR');let pct = totalSeg>0? Math.round(((totalSeg-restoSeg)/totalSeg)*100) : 0;document.getElementById('candlePct').textContent = pct+"%";}

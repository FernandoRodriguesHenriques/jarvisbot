<script>
let ATIVO=localStorage.getItem('JARVIS_ATIVO')||'XAUUSDm';
let MODO=localStorage.getItem('JARVIS_MODO')||'conservadora';
let BOT_ON=true;
let tvWidget=null;
document.getElementById('modo').value=MODO;
function tvSymbol(mt5sym){const map={"XAUUSDm":"OANDA:XAUUSD","BTCUSDm":"BINANCE:BTCUSD","EURUSDm":"FX:EURUSD","GBPUSDm":"FX:GBPUSD","US30m":"FOREXCOM:US30","NAS100m":"NASDAQ:NDX","USDJPYm":"FX:USDJPY"};return map[mt5sym]||"OANDA:XAUUSD";}
function carregarGrafico(){document.getElementById('tv_chart').innerHTML="";tvWidget = new TradingView.widget({"autosize": true,"symbol": tvSymbol(ATIVO),"interval": "1","timezone": "America/Sao_Paulo","theme": "dark","style": "1","locale": "br","backgroundColor": "#000000","gridColor": "rgba(42, 46, 57, 0.3)","container_id": "tv_chart","hide_top_toolbar": false,"hide_legend": false,"allow_symbol_change": false,"studies": ["EMA 9","EMA 21"]});}
function trocarModo(){MODO=document.getElementById('modo').value;localStorage.setItem('JARVIS_MODO',MODO);fetch('/set_modo?modo='+MODO);}
function mudarAtivo(){ATIVO=document.getElementById('ativoSel').value;localStorage.setItem('JARVIS_ATIVO',ATIVO); carregarGrafico();}
function toggleBot(){fetch('/bot/toggle').then(r=>r.json()).then(d=>{BOT_ON=d.ativo; atualizarBotUI();});}
function atualizarBotUI(){let btn=document.getElementById('botBtn');let txt=document.getElementById('botStatusText');let callBtn=document.getElementById('btnCall');let putBtn=document.getElementById('btnPut');let msg=document.getElementById('bloqueioMsg');if(BOT_ON){btn.textContent='ON';btn.className='btn-toggle on';txt.textContent='LIGADO';txt.style.color='#00c853';callBtn.disabled=false;putBtn.disabled=false;msg.style.display='none';}else{btn.textContent='OFF';btn.className='btn-toggle off';txt.textContent='DESLIGADO';txt.style.color='#ff3d57';callBtn.disabled=true;putBtn.disabled=true;msg.textContent='⛔ Bot DESLIGADO';msg.style.display='block';}}
function carregarBotStatus(){fetch('/bot/status').then(r=>r.json()).then(d=>{BOT_ON=d.ativo; atualizarBotUI();});}
function carregarAtivos(){fetch('/assets').then(r=>r.json()).then(lista=>{let sel=document.getElementById('ativoSel');sel.innerHTML="";lista.forEach(s=>{let o=document.createElement('option');o.value=s;o.textContent=s;if(s===ATIVO)o.selected=true;sel.appendChild(o);});document.getElementById('ativosCount').textContent=lista.length;});}
function carregarBanca(){fetch('/account').then(r=>r.json()).then(c=>{document.getElementById('bancaValor').textContent=`$ ${c.balance.toLocaleString('pt-BR',{minimumFractionDigits:2})}`;document.getElementById('equityValor').textContent=`$ ${c.equity.toLocaleString('pt-BR',{minimumFractionDigits:2})}`;let p=document.getElementById('profitValor');p.textContent=(c.profit>=0?'+':'')+c.profit.toFixed(2)+' USD';p.style.color=c.profit>=0?'#00c853':'#ff3d57';document.getElementById('margemValor').textContent=c.margin_level?c.margin_level.toFixed(2)+'%':'--%';document.getElementById('freeValor').textContent=`$ ${c.free_margin.toLocaleString('pt-BR',{minimumFractionDigits:2})}`;document.getElementById('accountHeader').textContent=`$ ${c.equity.toFixed(2)} | ${c.margin_level?c.margin_level.toFixed(0):0}% margem | ${c.profit>=0?'+':''}${c.profit.toFixed(2)} USD`;});}
function carregarScanner(){fetch('/analysis_all').then(r=>r.json()).then(lista=>{let html="";lista.forEach(a=>{let cor=a.guard>=80?'#00c853':a.guard>=60?'#ff9800':'#ff3d57';let lib=a.liberado?'✅':'❌';html+=`<div class="report-item"><span>${lib} ${a.symbol} <span style="color:${cor}">${a.guard}</span></span><span>${a.spread}pts ${a.guard_status}</span></div>`;});document.getElementById('scanner').innerHTML=html;});}
function loopAnalysis(){fetch(`/analysis?symbol=${ATIVO}&tf=1`).then(r=>r.json()).then(d=>{document.getElementById('preco').textContent=d.price;document.getElementById('spread').textContent=d.spread+'pts';document.getElementById('score').textContent=d.guard+'/100';document.getElementById('patterns').textContent=d.patterns && d.patterns.length>0? d.patterns.join(',') : 'Nenhum dos 12';document.getElementById('signal').textContent=d.signal;document.getElementById('conf').textContent=d.confidence+'%';document.getElementById('trades').textContent=(d.trades_hoje||0)+'/15';let badge=document.getElementById('badge');if(d.guard_status==='SEGURO'){badge.textContent='SEGURO';badge.style.background='#00c853';}else if(d.guard_status==='LIBERADO'){badge.textContent='LIBERADO';badge.style.background='#00c853';}else if(d.reason && d.reason.includes('BLOQUEADO')){badge.textContent='BLOQUEADO';badge.style.background='#ff3d57';}else{badge.textContent='ATENCAO';badge.style.background='#ff9800';}});}
function carregarPosicoes(){fetch('/positions').then(r=>r.json()).then(lista=>{document.getElementById('posCount').textContent=lista.length;let html="";if(lista.length==0)html="<span style='color:#666;font-size:9px'>Nenhuma - pronto para começar do zero</span>";else lista.forEach(p=>{let cor=p.profit>=0?'#00c853':'#ff3d57';html+=`<div class="pos-item"><span>${p.time} ${p.symbol} ${p.type.toUpperCase()} ${p.volume} <span style="color:${cor}">${p.profit>=0?'+':''}${p.profit} USD</span></span><button class="btn-fechar" onclick="fecharOrdem(${p.ticket})">X FECHAR</button></div>`;});document.getElementById('ordens').innerHTML=html;});}
function fecharOrdem(ticket){if(!confirm(`Fechar ${ticket}?`))return;fetch('/close',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({ticket})}).then(r=>r.json()).then(d=>{if(d.ok){carregarPosicoes();carregarBanca();carregarBrain();}else alert(d.error);});}
function resetarBrain(){
  if(!confirm("⚠️ RESETAR BRAIN? Apagar histórico, zerar 12 padrões, voltar para NORMAL. Banca continua. Começar do zero?")) return;
  fetch('/brain/reset',{method:'POST'}).then(r=>r.json()).then(d=>{
    if(d.ok){alert("✅ Brain resetado! Começando do zero"); carregarBrain(); carregarBanca();}
    else alert("Erro: "+d.error);
  });
}
function fecharTodas(){if(!confirm("Fechar TODAS?"))return;fetch('/positions').then(r=>r.json()).then(lista=>{lista.forEach(p=>{fetch('/close',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({ticket:p.ticket})}).then(()=>{carregarPosicoes();carregarBanca();});});});}
function enviarTrade(action){if(!BOT_ON){alert("Bot DESLIGADO"); return;} let lot=document.getElementById('lote').value;fetch('/trade',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({symbol:ATIVO,action,lot})}).then(r=>r.json()).then(d=>{if(!d.ok)alert(d.error);else{carregarPosicoes();carregarBanca();}});}
function carregarBrain(){fetch('/brain/status').then(r=>r.json()).then(b=>{let el=document.getElementById('brainEstado');el.textContent=b.estado; el.style.background=b.estado==='PROTECTION'?'#ff3d57':b.estado==='CAUTION'?'#ff9800':'#00c853';});fetch('/brain/trades').then(r=>r.json()).then(lista=>{let html="";lista.forEach(t=>{let cor=t.result==='WIN'?'#00c853':'#ff3d57';html+=`<div class="report-item"><span>${t.symbol} ${t.pattern}</span><span style="color:${cor}">${t.result} ${t.profit}</span></div>`;});document.getElementById('hist').innerHTML=html||'Brain zerado - aguardando primeiro trade';});fetch('/report/12').then(r=>r.json()).then(rep=>{let keys=Object.keys(rep);let html="";if(keys.length===0)html="Sem trades ainda - 12 padrões monitorados. Depois da primeira operação que você fechar com X FECHAR, ele já atualiza.";else{keys.forEach(pat=>{html+=`<div style="font-weight:800;margin-top:6px;color:#2962ff">${pat}</div>`;rep[pat].forEach(s=>{html+=`<div class="report-item"><span>${s.symbol}</span><span>${s.trades} | ${s.wins}W ${s.losses}L | ${s.win_rate}%</span></div>`;});});}document.getElementById('report12').innerHTML=html;});}
carregarAtivos(); carregarBotStatus(); carregarGrafico();
setInterval(carregarBanca,1000);
setInterval(carregarScanner,3000);
setInterval(loopAnalysis,2000);
setInterval(carregarPosicoes,2000);
setInterval(carregarBrain,5000);
carregarBanca();carregarScanner();loopAnalysis();carregarPosicoes();carregarBrain();
</script>

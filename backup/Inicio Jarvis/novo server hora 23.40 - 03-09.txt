def init_db():
    import sqlite3
    con = sqlite3.connect(DB)
    cur = con.cursor()
    cur.execute("CREATE TABLE IF NOT EXISTS trade_memory (id INTEGER PRIMARY KEY AUTOINCREMENT, ticket INTEGER, symbol TEXT, pattern TEXT, direction TEXT, entry REAL, exit REAL, volume REAL, profit REAL, result TEXT, timestamp TEXT)")
    cur.execute("CREATE TABLE IF NOT EXISTS pattern_stats (pattern TEXT, symbol TEXT, trades INTEGER, wins INTEGER, losses INTEGER, win_rate REAL, PRIMARY KEY (pattern, symbol))")
    for pat in PADROES_12:
        for sym in ATIVOS:
            cur.execute("INSERT OR IGNORE INTO pattern_stats VALUES (?,?,?,?,?,?)",(pat,sym,0,0,0,0.0))
    con.commit();con.close()

def pode_operar(symbol):
    global TRADES_HOJE, DATA_HOJE
    if not BOT_ATIVO:
        return False, "BLOQUEADO: Bot DESLIGADO"
    hoje = datetime.date.today().isoformat()
    if hoje!= DATA_HOJE:
        DATA_HOJE = hoje; TRADES_HOJE = 0; ULTIMO.clear()
    pos = mt5.positions_get()
    if pos and len(pos) >= MAX_TOTAL:
        return False, f"BLOQUEADO: Total {len(pos)}/{MAX_TOTAL}"
    if symbol:
        pos_sym = [p for p in (pos or []) if p.symbol == symbol]
        if len(pos_sym) >= MAX_POR_ATIVO:
            return False, f"BLOQUEADO: {symbol} ja tem {len(pos_sym)}"
        ultimo = ULTIMO.get(symbol,0)
        rest = COOLDOWN - (time.time()-ultimo)
        if ultimo!=0 and rest>0:
            return False, f"BLOQUEADO: Cooldown {int(rest)}s"
    conta = get_conta()
    if conta:
        if conta.profit <= STOP_DIARIO:
            return False, f"BLOQUEADO: Stop {conta.profit}"
        if conta.margin_level and conta.margin_level < 150:
            return False, f"BLOQUEADO: Margem {conta.margin_level:.0f}%"
    if TRADES_HOJE >= MAX_TRADES_DIA:
        return False, "BLOQUEADO: Max trades"
    return True, "LIBERADO"
def ema_calc(data, period):
    if len(data) < period: return data[-1] if data else 0
    k = 2/(period+1)
    e = sum(data[:period])/period
    for p in data[period:]: e = p*k + e*(1-k)
    return e
def detect_patterns(closes):
    if len(closes) < 5: return []
    o=closes[-2];c=closes[-1];h=max(closes[-3:]);l=min(closes[-3:])
    body=abs(c-o);amp=h-l if h!=l else 0.0001
    res=[]
    if body < amp*0.3 and (o-l)>body*1.5: res.append("HAMMER")
    if body < amp*0.1: res.append("DOJI")
    if c>o and closes[-2]<closes[-3]: res.append("ENGULFING_BULL")
    if c<o and closes[-2]>closes[-3]: res.append("ENGULFING_BEAR")
    if body > amp*0.8: res.append("MARUBOZU")
    return res
def executar_ordem(symbol, action, lot):
    global TRADES_HOJE
    liberado,msg=pode_operar(symbol)
    if not liberado:
        print(f"NEGADO {symbol} {msg}")
        return None, msg
    mt5.symbol_select(symbol, True)
    tick=mt5.symbol_info_tick(symbol)
    if not tick:
        return None, "Sem tick"
    price=tick.ask if action=='buy' else tick.bid
    otype=mt5.ORDER_TYPE_BUY if action=='buy' else mt5.ORDER_TYPE_SELL
    req={"action":mt5.TRADE_ACTION_DEAL,"symbol":symbol,"volume":lot,"type":otype,"price":price,"deviation":100,"magic":6502026,"comment":f"JARVIS AUTO {MODO_ATUAL}","type_time":mt5.ORDER_TIME_GTC,"type_filling":mt5.ORDER_FILLING_IOC}
    res=mt5.order_send(req)
    if not res or res.retcode!=mt5.TRADE_RETCODE_DONE:
        req["type_filling"]=mt5.ORDER_FILLING_RETURN
        res=mt5.order_send(req)
    if res and res.retcode==mt5.TRADE_RETCODE_DONE:
        TRADES_HOJE+=1
        ULTIMO[symbol]=time.time()
        print(f"OK ORDEM {symbol} {action} {lot} ticket {res.order}")
        return res, "OK"
    err = res.comment if res else "erro desconhecido"
    print(f"FALHA {symbol} {action} {err}")
    return None, err
def loop_robo():
    print("ROBO AUTO EMA 9/21 INICIADO - ALTERNANDO 7 ATIVOS")
    while True:
        try:
            if not BOT_ATIVO:
                time.sleep(2)
                continue
            pos = mt5.positions_get()
            total = len(pos) if pos else 0
            if total >= MAX_TOTAL:
                time.sleep(3)
                continue
            for sym in ATIVOS:
                if not BOT_ATIVO: break
                mt5.symbol_select(sym, True)
                tick = mt5.symbol_info_tick(sym)
                info = mt5.symbol_info(sym)
                if not tick or not info: continue
                spread = int((tick.ask - tick.bid)/info.point) if info.point else 999
                limite = SPREAD_LIMITES[MODO_ATUAL]
                guard = 100 if spread <= limite else 60 if MODO_ATUAL=="conservadora" else 75
                if guard < 60: continue
                key = f"{sym}_1"
                if key not in price_history: price_history[key]=deque(maxlen=200)
                price_history[key].append(tick.bid)
                closes=list(price_history[key])
                if len(closes)<30: continue
                ema9=ema_calc(closes,9)
                ema21=ema_calc(closes,21)
                patterns=detect_patterns(closes)
                if not patterns: continue
                liberado,msg=pode_operar(sym)
                if not liberado: continue
                action = 'buy' if ema9>ema21 else 'sell'
                print(f"SINAL EMA {sym} {action} EMA9 {ema9:.2f} EMA21 {ema21:.2f} {patterns} guard {guard}")
                lot = 0.10 if MODO_ATUAL=="conservadora" else 0.20
                executar_ordem(sym, action, lot)
                time.sleep(2)
            time.sleep(3)
        except Exception as e:
            print(f"ERRO LOOP {e}")
            time.sleep(3)
threading.Thread(target=loop_robo, daemon=True).start()

@app.route('/brain/status')
def brain_status():
    import sqlite3
    con=sqlite3.connect(DB)
    cur=con.cursor()
    cur.execute("SELECT COUNT(*), SUM(CASE WHEN result='WIN' THEN 1 ELSE 0 END), SUM(CASE WHEN result='LOSS' THEN 1 ELSE 0 END) FROM trade_memory")
    total,wins,losses=cur.fetchone()
    total=total or 0;wins=wins or 0;losses=losses or 0
    wr=(wins/total*100) if total>0 else 0
    con.close()
    estado="NORMAL"
    if losses>=3: estado="PROTECTION"
    elif losses>=2: estado="CAUTION"
    elif wins>=3: estado="PERFORMANCE"
    return jsonify({"total":total,"wins":wins,"losses":losses,"win_rate":round(wr,2),"estado":estado})

def report12():
    import sqlite3
    con=sqlite3.connect(DB)
    cur=con.cursor()
    cur.execute("SELECT pattern,symbol,trades,wins,losses,win_rate FROM pattern_stats WHERE trades>0")
    rows=cur.fetchall();con.close()
    report={}
    for pat,sym,tr,w,l,wr in rows:
        if pat not in report: report[pat]=[]
        report[pat].append({"symbol":sym,"trades":tr,"wins":w,"losses":l,"win_rate":wr})

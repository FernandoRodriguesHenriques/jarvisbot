<script>
let currentSymbol='EURUSDm', currentTF='1', botActive=false, botTradeCount=0, lastBotTrade=0, currentSignal='AGUARDAR', currentConf=0;
function getTVSymbol(s){return s.replace('m','').replace('M','');}
function changeAsset(){currentSymbol=document.getElementById('asset').value; loadChart();}
function setTF(tf,btn){currentTF=tf; document.querySelectorAll('.tf-btn').forEach(b=>b.classList.remove('active')); btn.classList.add('active'); loadChart();}
function loadChart(){
  document.getElementById('tradingview_widget').innerHTML='';
  let sym=getTVSymbol(currentSymbol);
  let map={'EURUSD':'FX:EURUSD','GBPUSD':'FX:GBPUSD','USDJPY':'FX:USDJPY','AUDUSD':'FX:AUDUSD','BTCUSD':'BINANCE:BTCUSD','XAUUSD':'OANDA:XAUUSD','US500':'OANDA:SPX500USD'};
  new TradingView.widget({"autosize":true,"symbol":map[sym]||'FX:'+sym,"interval":currentTF,"timezone":"America/Sao_Paulo","theme":"dark","style":"1","locale":"br","container_id":"tradingview_widget"});
}
document.getElementById('lote').addEventListener('change',e=>{let v=e.target.value*100; document.getElementById('callVal').innerText='$'+v; document.getElementById('putVal').innerText='$'+v;});
function toggleBot(){
  botActive=!botActive;
  let btn=document.getElementById('botBtn');
  if(botActive){btn.innerText='ON ●'; btn.className='bot-toggle bot-on'; log('BOT LIGADO!');}
  else{btn.innerText='OFF'; btn.className='bot-toggle bot-off'; log('BOT DESLIGADO'); document.getElementById('botCooldown').innerText='';}
}
async function updateSaldo(){try{let r=await fetch('http://127.0.0.1:5000/balance'); let d=await r.json(); if(d.ok)document.getElementById('saldoTop').innerText='$ '+d.balance.toFixed(2)+' | L: '+d.profit.toFixed(2);}catch(e){}}
async function updatePositions(){try{let r=await fetch('http://127.0.0.1:5000/positions'); let d=await r.json(); if(d.ok){let h=''; if(d.positions.length==0)h='Nenhuma'; else d.positions.forEach(p=>{let c=p.profit>=0?'#00c853':'#ff1744'; h+=<div class="pos-item"><div><b>${p.symbol}</b> ${p.type.toUpperCase()}<br><span style="color:${c}">$${p.profit.toFixed(2)}</span></div><button class="close-btn" onclick="closePos(${p.ticket})">FECHAR</button></div>}); document.getElementById('posList').innerHTML=h; return d.positions.length;} }catch(e){} return 0}
async function updateAnalysis(){
  try{
    let r=await fetch('http://127.0.0.1:5000/analysis?symbol='+currentSymbol+'&tf='+currentTF);
    let d=await r.json();
    if(d.ok){
      document.getElementById('a_price').innerText=d.price.toFixed(5);
      document.getElementById('a_ema').innerText=d.ema9.toFixed(5)+' / '+d.ema21.toFixed(5);
      document.getElementById('a_rsi').innerText=d.rsi;
      document.getElementById('a_macd').innerText=d.macd+' / '+d.macd_signal;
      document.getElementById('a_bb').innerText=d.bb_lower.toFixed(5)+' - '+d.bb_upper.toFixed(5);
      let s=document.getElementById('a_signal'); s.className='signal '+d.sinal; s.innerText=d.sinal;
      document.getElementById('a_conf').innerText=d.confianca+'%';
      document.getElementById('a_detalhes').innerText=d.detalhes;
      currentSignal=d.sinal; currentConf=d.confianca;
      // BOT LOGIC
      if(botActive){ checkBot(); }
    }
  }catch(e){}
}
async function checkBot(){
  let confMin=parseInt(document.getElementById('botConf').value);
  let now=Date.now()/1000;
  let cooldown=120; // 2 min
  if(now-lastBotTrade < cooldown){
    let rest=Math.ceil(cooldown-(now-lastBotTrade));
    document.getElementById('botCooldown').innerText=rest+'s cooldown';
    return;
  }
  document.getElementById('botCooldown').innerText='PRONTO';
  if(currentSignal=='AGUARDAR' || currentConf < confMin) return;
  let posCount=await updatePositions();
  if(posCount>0) { document.getElementById('botCooldown').innerText='Com ordem aberta'; return; } // só 1 por vez
  // EXECUTA BOT
  let lote=document.getElementById('botLote').value;
  let tipo = currentSignal=='CALL'? 'buy' : 'sell';
  log('BOT: '+currentSignal+' '+currentConf+'%');
  let res=await fetch('http://127.0.0.1:5000/trade',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({action:tipo,symbol:currentSymbol,lot:parseFloat(lote)})});
  let d=await res.json();
  if(d.ok){
    botTradeCount++; lastBotTrade=now;
    document.getElementById('botTrades').innerText=botTradeCount;
    document.getElementById('botLast').innerText=currentSignal+' '+currentConf+'%';
    log('BOT ABRIU '+d.ticket);
    updatePositions();
  }
}
async function trade(t){let l=document.getElementById('lote').value; log('Manual...'); let r=await fetch('http://127.0.0.1:5000/trade',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({action:t,symbol:currentSymbol,lot:parseFloat(l)})}); let d=await r.json(); if(d.ok){log('ABERTA '+d.ticket); updatePositions(); updateSaldo();}else{log(d.error);}}
async function closePos(t){if(!confirm('Fechar '+t+'?'))return; let r=await fetch('http://127.0.0.1:5000/close',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({ticket:t})}); let d=await r.json(); if(d.ok){log('Fechada'); updatePositions(); updateSaldo();}}
function log(m){document.getElementById('log').innerText=m;}
loadChart(); setInterval(()=>{updateSaldo(); updatePositions(); updateAnalysis();},2500); updateSaldo(); updatePositions(); updateAnalysis();
</script>

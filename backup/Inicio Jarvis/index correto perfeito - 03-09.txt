<script>
let currentTF='1', trades=0, lastTrade=0;
function loadTV(sym='XAUUSDm'){
 let map={'XAUUSDm':'OANDA:XAUUSD','EURUSDm':'OANDA:EURUSD','GBPUSDm':'OANDA:GBPUSD','USDJPYm':'OANDA:USDJPY','BTCUSDm':'COINBASE:BTCUSD','US30m':'OANDA:US30USD','NAS100m':'NASDAQ:NDX'};
 let tvSym=map[sym]||'OANDA:XAUUSD'; document.getElementById('tv').innerHTML='';
 new TradingView.widget({"autosize":true,"symbol":tvSym,"interval":currentTF,"timezone":"America/Sao_Paulo","theme":"dark","style":"1","locale":"br","container_id":"tv","hide_top_toolbar":false,"withdateranges":false});
}
function setTF(tf){currentTF=tf; document.querySelectorAll('.tf').forEach(b=>b.classList.toggle('active',b.dataset.tf==tf)); loadTV(document.getElementById('sym').value);}
async function doTrade(tipo){
 let sym=document.getElementById('sym').value; let lote=parseFloat(document.getElementById('loteManual').value)||0.10;
 try{
  let r=await fetch('/trade',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({symbol:sym,action:tipo,lot:lote})});
  let j=await r.json();
  if(j.ok){trades++; document.getElementById('tradesHoje').innerText=trades; document.getElementById('ordens').innerHTML=`<div style="color:#00ff88">${new Date().toLocaleTimeString()} ✅ ${tipo.toUpperCase()} ${sym} Ticket ${j.ticket}</div>`+document.getElementById('ordens').innerHTML;}
  else{ alert(j.error); }
 }catch(e){alert(e);}
}
async function upd(){
 try{
  let sym=document.getElementById('sym').value;
  let r=await fetch('/analysis?symbol='+sym+'&tf='+currentTF+'&t='+Date.now());
  let d=await r.json();
  if(!d.price) return;
  document.getElementById('aPreco').innerText=d.price;
  document.getElementById('aEma').innerText=d.ema9+' / '+d.ema21;
  document.getElementById('aRsi').innerText=d.rsi;
  document.getElementById('aMacd').innerText=d.macd;
  document.getElementById('aBb').innerText=d.bb_low+' / '+d.bb_up;
  document.getElementById('aSignal').innerText=d.signal;
  document.getElementById('aSignal').className='signal '+d.signal;
  document.getElementById('aConf').innerText=d.confidence+'%';
  document.getElementById('aDet').innerText=d.reason||d.collect;
  document.getElementById('guardScore').innerText=d.guard+' /100';
  let st=document.getElementById('guardStatus');
  st.innerText=d.guard_status;
  if(d.guard_status=='ATENÇÃO'){st.className='badge badge-aten';}
  else if(d.guard>=80){st.className='badge badge-on'; st.innerText='SEGURO';}
  else{st.className='badge badge-off';}
  document.getElementById('guardSpread').innerText=d.spread+' pts (lim 80)';
  document.getElementById('guardMotivo').innerText=d.guard_msg;
 }catch(e){}
}
setInterval(async()=>{
 let bot=document.getElementById('bot');
 let st=document.getElementById('botStatus');
 if(!bot.checked){st.innerText='OFF'; st.className='badge badge-off'; return;}
 st.innerText='ON'; st.className='badge badge-on';
 let signal=document.getElementById('aSignal').innerText.trim();
 let conf=parseInt(document.getElementById('aConf').innerText)||0;
 let guard=parseInt(document.getElementById('guardScore').innerText)||0;
 let confMin=parseInt(document.getElementById('botConf').value)||80;
 if((signal=='CALL'||signal=='PUT')&&conf>=confMin&&guard>=60&&Date.now()-lastTrade>120000){
  await doTrade(signal=='CALL'?'buy':'sell'); lastTrade=Date.now();
 }
},2500);
document.getElementById('sym').addEventListener('change',e=>{loadTV(e.target.value);});
loadTV('XAUUSDm'); setInterval(upd,2000); upd();
</script>
